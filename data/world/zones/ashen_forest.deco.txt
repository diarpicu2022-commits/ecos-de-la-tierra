; Leyenda: carácter = tipo, id, sólido|pisable[, ancho x alto en tiles]
; Las viñetas son de dos elementos que el jugador une; su número es el orden

; 1 = vignette, v1_stump, sólido              tala: tocón de corte limpio
; a = vignette, v1_felled_log, sólido, 2x1    tala: el tronco talado

; 2 = vignette, v2_stump, sólido              el claro pelado que dejan
; q = vignette, v3_burnt_tree, sólido, 2x2    el fuego: el primer árbol quemado
; 3 = vignette, v3_ash, pisable               la ceniza, en diagonal hacia la fuente
; Y = landmark, burnt_oak, sólido, 3x4        hito: el roble quemado mayor, en la fuente
; M = entity, monster, sólido                 la Llama de la Deforestación (combate de historia)
; X = entity, blocker, sólido                 paso cerrado: troncos en brasas, detrás de la fuente
; S = entity, seedbed, sólido                 semillero del ramal norte
; T = decor, tree, sólido, 2x2                árbol enfermo, en pie
; U = decor, tree_b, sólido, 2x2              árbol enfermo, segunda copa
; c = decor, burnt_tree, sólido, 2x2          árbol quemado (el gradiente)
; t = decor, stump, sólido                    tocón
; e = decor, ash, pisable                     ceniza
; g = decor, grass, pisable                   hierba alta
; R = decor, rock, sólido                     roca partida
; @ = spawn, start, pisable                   llegada desde Valdehoja
